\begin{frame}
	\frametitle{Ferramentas de Análise Vetorial}
	\begin{table}[h]
	\centering
	\scriptsize 
	\begin{tabular}{|l|p{4cm}|c|}
		\hline
		\textbf{Relação Espacial} & \textbf{Descrição} & \textbf{DE-9IM} \\ \hline
		\textbf{Disjunto (Disjoint)} & Os objetos não se tocam nem se cruzam. & \texttt{FF*FF****} \\ \hline
		\textbf{Toca (Touches)} & Os objetos encostam apenas na borda (fronteira). & \texttt{FT*******} ou \texttt{F**T*****} \\ \hline
		\textbf{Sobrepõe (Overlaps)} & Os objetos têm parte em comum, mas nenhum está totalmente dentro do outro. & \texttt{T*T***T**} \\ \hline
		\textbf{Dentro (Within)} & O primeiro objeto está completamente dentro do segundo. & \texttt{T*F**F***} \\ \hline
		\textbf{Contém (Contains)} & O primeiro objeto contém totalmente o segundo. & \texttt{T*****FF*} \\ \hline
		\textbf{Igual (Equals)} & Os dois objetos são exatamente iguais (mesmo tamanho e localização). & \texttt{TFFFTFFFT} \\ \hline
		\textbf{Interseção (Intersects)} & Qualquer tipo de contato entre os objetos. & Não disjunto \\ \hline
	\end{tabular}
	\caption{Relações espaciais básicas representadas com a matriz DE-9IM.}
	\end{table}
\end{frame}
